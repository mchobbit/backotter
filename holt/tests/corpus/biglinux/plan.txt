# ---- partition ----
# vda -> /dev/sdq: clear the old table and signatures
wipefs --all /dev/sdq
# /dev/sdq: its first 4 MiB zeroed, and its last 1 MiB (a GPT's backup, md 0.90/1.0 superblocks, ZFS's end labels)
dd if=/dev/zero of=/dev/sdq bs=1M count=4 conv=fsync
sh -c 'n=$(blockdev --getsz /dev/sdq) && dd if=/dev/zero of=/dev/sdq bs=512 seek=$((n - 2048)) count=2048 conv=fsync'
# the new partitions' old content zeroed through /dev/sdq, before its table names them (blocks of 512 bytes, at their start= sectors)
# vda1: its first 4 MiB and its last 1 MiB
dd if=/dev/zero of=/dev/sdq bs=512 seek=2048 count=8192 conv=fsync
dd if=/dev/zero of=/dev/sdq bs=512 seek=614400 count=2048 conv=fsync
# vda2: its first 4 MiB and its last 1 MiB
dd if=/dev/zero of=/dev/sdq bs=512 seek=616448 count=8192 conv=fsync
dd if=/dev/zero of=/dev/sdq bs=512 seek=50327552 count=2048 conv=fsync
# udev done with /dev/sdq before sfdisk takes it
udevadm settle
sfdisk --wipe always --wipe-partitions always /dev/sdq <<'EOF'

label-id: bc000000-0000-4000-8000-000000000001

# ---- format ----
# vda1: vfat "EFI"
mkfs.fat -F 32 -i BC000001 -n EFI -S 512 -s 8 -f 2 -R 32 -h 2048 /dev/sdq1
# check: /dev/sdq1 is vfat with the source's FAT size, sector and cluster size (read back by bko-holt's own probe)
# vda2: btrfs
mkfs.btrfs -f -O extref,skinny-metadata,no-holes,^raid56,^raid1c34,^zoned,^squota,free-space-tree,block-group-tree -U bc000000-0000-4000-8000-000000000004 --csum crc32c -n 16384 -s 4096 -d single -m dup --device-uuid bc000000-0000-4000-8000-000000000005 /dev/sdq2
# check: /dev/sdq2 is btrfs with the source's features (read back by bko-holt's own probe)
mkdir -p /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000004
# btrfs bc000000-0000-4000-8000-000000000004: its subvolumes
mount -t btrfs -o subvolid=5 /dev/sdq2 /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000004
btrfs subvolume create /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000004/@
btrfs subvolume create /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000004/@home
btrfs subvolume create /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000004/@swapfile
# btrfs bc000000-0000-4000-8000-000000000004: the default stays the top level (ID 5), set as on the source, whose superblock has the default_subvol flag that only set-default sets
btrfs subvolume set-default /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000004
umount /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000004

# ---- mount ----
mkdir -p /mnt/bko-target
mount -t btrfs -o subvol=/@,noatime,compress-force=zstd,autodefrag /dev/sdq2 /mnt/bko-target
mkdir -p /mnt/bko-target/home

# ---- restore ----
