# bko-holt load plan for layout.json
# Source: host-1, Debian GNU/Linux 13 (trixie), BIOS boot with grub, dumped 2026-11-08T02:34:17Z
#
# Disks:
#   vda (16 GiB) -> /dev/sdq
#
# Notes:
#   - debconf's grub-pc/install_devices still names the source's disks; `dpkg-reconfigure grub-pc` on the restored system fixes it

# ---- partition ----
# vda -> /dev/sdq: clear the old table and signatures
wipefs --all /dev/sdq
# /dev/sdq: its first 4 MiB zeroed, and its last 1 MiB (a GPT's backup, md 0.90/1.0 superblocks, ZFS's end labels)
dd if=/dev/zero of=/dev/sdq bs=1M count=4 conv=fsync
sh -c 'n=$(blockdev --getsz /dev/sdq) && dd if=/dev/zero of=/dev/sdq bs=512 seek=$((n - 2048)) count=2048 conv=fsync'
# the new partitions' old content zeroed through /dev/sdq, before its table names them (blocks of 512 bytes, at their start= sectors)
# vda1: its first 4 MiB and its last 1 MiB
dd if=/dev/zero of=/dev/sdq bs=512 seek=2048 count=8192 conv=fsync
dd if=/dev/zero of=/dev/sdq bs=512 seek=2097152 count=2048 conv=fsync
# vda2: its first 4 MiB and its last 1 MiB
dd if=/dev/zero of=/dev/sdq bs=512 seek=2099200 count=8192 conv=fsync
dd if=/dev/zero of=/dev/sdq bs=512 seek=6291456 count=2048 conv=fsync
# vda3: its first 4 MiB and its last 1 MiB
dd if=/dev/zero of=/dev/sdq bs=512 seek=6293504 count=8192 conv=fsync
dd if=/dev/zero of=/dev/sdq bs=512 seek=33552384 count=2048 conv=fsync
# udev done with /dev/sdq before sfdisk takes it
udevadm settle
sfdisk --wipe always --wipe-partitions always /dev/sdq <<'EOF'
label: dos
label-id: 0xbc000001
unit: sectors
sector-size: 512

/dev/sdq1 : start=2048, size=2097152, type=83, bootable
/dev/sdq2 : start=2099200, size=4194304, type=82
/dev/sdq3 : start=6293504, size=27260928, type=bf
EOF
udevadm settle

# ---- assemble ----
# ZFS pool rpool, with the source's feature flags only
zpool create -f -d -o cachefile=none -o ashift=12 -o autotrim=on -o feature@async_destroy=enabled -o feature@empty_bpobj=enabled -o feature@lz4_compress=enabled -o feature@multi_vdev_crash_dump=enabled -o feature@spacemap_histogram=enabled -o feature@enabled_txg=enabled -o feature@hole_birth=enabled -o feature@extensible_dataset=enabled -o feature@embedded_data=enabled -o feature@bookmarks=enabled -o feature@filesystem_limits=enabled -o feature@large_blocks=enabled -o feature@large_dnode=enabled -o feature@sha512=enabled -o feature@skein=enabled -o feature@edonr=enabled -o feature@userobj_accounting=enabled -o feature@encryption=enabled -o feature@project_quota=enabled -o feature@device_removal=enabled -o feature@obsolete_counts=enabled -o feature@zpool_checkpoint=enabled -o feature@spacemap_v2=enabled -o feature@allocation_classes=enabled -o feature@resilver_defer=enabled -o feature@bookmark_v2=enabled -o feature@redaction_bookmarks=enabled -o feature@redacted_datasets=enabled -o feature@bookmark_written=enabled -o feature@log_spacemap=enabled -o feature@livelist=enabled -o feature@device_rebuild=enabled -o feature@zstd_compress=enabled -o feature@draid=enabled -o feature@zilsaxattr=enabled -o feature@head_errlog=enabled -o feature@blake3=enabled -o feature@block_cloning=enabled -o feature@vdev_zaps_v2=enabled -o feature@redaction_list_spill=enabled -o feature@raidz_expansion=enabled -o feature@fast_dedup=enabled -o feature@longname=enabled -o feature@large_microzap=enabled -O mountpoint=/ -O compression=lz4 -O canmount=off -O xattr=on -O dnodesize=auto -O acltype=posix -O relatime=on -O normalization=formD -O utf8only=on -R /mnt/bko-target rpool /dev/sdq3
# the source pool's GUID (vdev GUIDs can't be kept)
zpool reguid -g 10000000000000000001 rpool
# mount mounts every dataset in order
sh -c '[ "$(zfs get -H -o value mounted rpool)" != yes ] || zfs unmount rpool'

# ---- format ----
# vda1: ext4
mke2fs -F -q -t ext4 -O none,has_journal,ext_attr,resize_inode,dir_index,orphan_file,filetype,extent,64bit,flex_bg,metadata_csum_seed,sparse_super,large_file,huge_file,dir_nlink,extra_isize,metadata_csum -U bc000000-0000-4000-8000-000000000001 -b 4096 -I 256 -N 65536 -g 32768 -G 16 -E hash_seed=bc000000-0000-4000-8000-000000000002,desc_size=64,resize=268435456 /dev/sdq1 1048576k
tune2fs -r 13107 /dev/sdq1
# vda2: swap
mkswap -U bc000000-0000-4000-8000-000000000003 --pagesize 4096 /dev/sdq2
zfs create -u -o mountpoint=none -o canmount=off -o normalization=formD -o utf8only=on rpool/ROOT
zfs create -u -o mountpoint=/ -o canmount=noauto -o normalization=formD -o utf8only=on rpool/ROOT/debian
zfs create -u -o normalization=formD -o utf8only=on rpool/home
zfs create -u -o mountpoint=/root -o normalization=formD -o utf8only=on rpool/home/root
zfs create -u -o normalization=formD -o utf8only=on rpool/srv
zfs create -u -o canmount=off -o normalization=formD -o utf8only=on rpool/usr
zfs create -u -o normalization=formD -o utf8only=on rpool/usr/local
zfs create -u -o canmount=off -o normalization=formD -o utf8only=on rpool/var
zfs create -u -o com.sun:auto-snapshot=false -o normalization=formD -o utf8only=on rpool/var/cache
zfs create -u -o canmount=off -o normalization=formD -o utf8only=on rpool/var/lib
zfs create -u -o normalization=formD -o utf8only=on rpool/var/log
zfs create -u -o normalization=formD -o utf8only=on rpool/var/mail
zfs create -u -o normalization=formD -o utf8only=on rpool/var/spool

# ---- mount ----
zfs mount rpool/ROOT/debian
mkdir -p /mnt/bko-target/boot
mount -t ext4 -o defaults /dev/sdq1 /mnt/bko-target/boot
zfs mount rpool/home
zfs mount rpool/home/root
zfs mount rpool/srv
zfs mount rpool/usr/local
zfs mount rpool/var/cache
zfs mount rpool/var/log
zfs mount rpool/var/mail
zfs mount rpool/var/spool

# ---- restore ----
# snapshot latest of /srv/bko-repo, with BackOtter's restore flags
RUSTIC_STRICT_EXIT=true rustic-bko --repository /srv/bko-repo --no-progress restore --numeric-id latest /mnt/bko-target
# the atimes the restore set, for boot to put back after its reads
sh -c 'mkdir -p /run/bko-holt && find /mnt/bko-target \( -path /mnt/bko-target/dev -o -path /mnt/bko-target/proc -o -path /mnt/bko-target/sys -o -path /mnt/bko-target/run \) -prune -o -printf '\''%A@\t%p\0'\'' | LC_ALL=C sort -z > /run/bko-holt/atimes.before'

# ---- boot ----
# refuses to start unless /mnt/bko-target/etc/fstab is the restored system's (its / matches the layout's)
# the restored system's zpool.cache names the new pool
zpool set cachefile=/mnt/bko-target/etc/zfs/zpool.cache rpool
# the restored atimes (the restore stage lists them), to put back after this stage's reads
sh -c '[ -s /run/bko-holt/atimes.before ] || { mkdir -p /run/bko-holt && find /mnt/bko-target \( -path /mnt/bko-target/dev -o -path /mnt/bko-target/proc -o -path /mnt/bko-target/sys -o -path /mnt/bko-target/run \) -prune -o -printf '\''%A@\t%p\0'\'' | LC_ALL=C sort -z > /run/bko-holt/atimes.before; }'
# the restored LVM devices file, kept
sh -c 'rm -rf /run/bko-holt/lvm-devices; [ ! -d /mnt/bko-target/etc/lvm/devices ] || { cp -a /mnt/bko-target/etc/lvm/devices /run/bko-holt/lvm-devices && touch -r /mnt/bko-target/etc/lvm /run/bko-holt/lvm-devices.parent; }'
mount --rbind --make-rslave /dev /mnt/bko-target/dev
mount --rbind --make-rslave /proc /mnt/bko-target/proc
mount --rbind --make-rslave /sys /mnt/bko-target/sys
mount --rbind --make-rslave /run /mnt/bko-target/run
chroot /mnt/bko-target update-initramfs -u -k all
chroot /mnt/bko-target grub-install --target=i386-pc /dev/sdq
# os-prober off: it would list the rescue system's disks
sh -c '[ ! -e /mnt/bko-target/usr/bin/os-prober ] || mount --bind /mnt/bko-target/usr/bin/true /mnt/bko-target/usr/bin/os-prober'
chroot /mnt/bko-target env LANG=C update-grub
sh -c '! grep -q '\'' /mnt/bko-target/usr/bin/os-prober '\'' /proc/self/mountinfo || umount /mnt/bko-target/usr/bin/os-prober'
mkdir -p /mnt/bko-target/var/log/bko-holt
# this restore's log, on the restored system
cp -a /run/bko-holt/log/. /mnt/bko-target/var/log/bko-holt/
# SELinux labels for the files the boot stage wrote
chroot /mnt/bko-target sh -c '[ -f /etc/selinux/config ] || exit 0; . /etc/selinux/config; [ "${SELINUX:-disabled}" != disabled ] || exit 0; f=/etc/selinux/${SELINUXTYPE:-targeted}/contexts/files/file_contexts; [ ! -f "$f" ] || ! command -v setfiles >/dev/null || { [ ! -d /etc/kernel ] || set -- /etc/kernel; setfiles "$f" /boot /var/log/bko-holt "$@"; }'
umount --recursive /mnt/bko-target/run
umount --recursive /mnt/bko-target/sys
umount --recursive /mnt/bko-target/proc
umount --recursive /mnt/bko-target/dev
# the restored LVM devices file, put back
sh -c '[ ! -d /run/bko-holt/lvm-devices ] || { rm -rf /mnt/bko-target/etc/lvm/devices && cp -a /run/bko-holt/lvm-devices /mnt/bko-target/etc/lvm/devices && touch -r /run/bko-holt/lvm-devices.parent /mnt/bko-target/etc/lvm; }'
# the atimes this stage's reads changed, put back
bash -c 'find /mnt/bko-target \( -path /mnt/bko-target/dev -o -path /mnt/bko-target/proc -o -path /mnt/bko-target/sys -o -path /mnt/bko-target/run \) -prune -o -printf '\''%A@\t%p\0'\'' | LC_ALL=C sort -z > /run/bko-holt/atimes.after && LC_ALL=C comm -z -23 /run/bko-holt/atimes.before /run/bko-holt/atimes.after | \
             while IFS=$'\''\t'\'' read -r -d '\'''\'' a p; do [ -L "$p" ] || [ -e "$p" ] || continue; touch -a -h -d "@$a" -- "$p" 2>/dev/null || { [ ! -L "$p" ] && lsattr -d -- "$p" 2>/dev/null | cut -d'\'' '\'' -f1 | grep -q i && { chattr -i -- "$p"; touch -a -h -d "@$a" -- "$p"; chattr +i -- "$p"; }; }; done'

# ---- close ----
umount --recursive /mnt/bko-target
# other mount namespaces' copies of the target's mounts
sh -c 'me=$(readlink /proc/self/ns/mnt); seen=" $me "; at() { awk -v t=/mnt/bko-target '\''{ m = $5; if (m == t || length(m) > length(t) && substr(m, length(m) - length(t) + 1) == t) print m }'\''; }; for f in /proc/[0-9]*/mountinfo; do grep -q '\''/mnt/bko-target[ /]'\'' "$f" 2>/dev/null || continue; p=${f#/proc/}; p=${p%/mountinfo}; ns=$(readlink /proc/$p/ns/mnt) || continue; case $seen in *" $ns "*) continue;; esac; seen="$seen$ns "; echo "unmounting /mnt/bko-target in $ns (pid $p, $(cat /proc/$p/comm))"; for m in $(at <"$f"); do nsenter -t "$p" -m umount --recursive "$m" || true; done; done; awk '\''{ for (i = 7; i < NF; i++) if ($i == "-") { if ($(i + 1) == "nsfs" && $4 ~ /^mnt:\[/) print $4, $5; break } }'\'' /proc/self/mountinfo | while read -r ns b; do case $seen in *" $ns "*) continue;; esac; seen="$seen$ns "; ms=$(nsenter --mount="$b" cat /proc/self/mountinfo | at); [ -n "$ms" ] || continue; echo "unmounting /mnt/bko-target in $ns (bound at $b)"; for m in $ms; do nsenter --mount="$b" umount --recursive "$m" || true; done; done'
sh -c 'i=0; until udevadm settle; zpool export rpool; do i=$((i + 1)); [ $i -lt 5 ] || exit 1; sleep 1; done'
