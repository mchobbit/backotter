mkfs.btrfs -f -O extref,skinny-metadata,no-holes,^raid56,^raid1c34,^zoned,^squota,free-space-tree,block-group-tree -U bc000000-0000-4000-8000-000000000006 -L fedora_host-1 --csum crc32c -n 16384 -s 4096 -d single -m dup --device-uuid bc000000-0000-4000-8000-000000000007 /dev/sdq3
mkdir -p /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000006
# btrfs bc000000-0000-4000-8000-000000000006: its subvolumes
mount -t btrfs -o subvolid=5 /dev/sdq3 /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000006
btrfs subvolume create /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000006/root
mkdir -p /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000006/root/var/lib
