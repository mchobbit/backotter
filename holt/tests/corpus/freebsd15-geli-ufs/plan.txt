# bko-holt load plan for layout.json
# Source: host-1, FreeBSD 15.1-RELEASE-p7, UEFI boot with freebsd, dumped 2026-11-17T23:19:46Z
#
# Disks:
#   vtbd0 (16 GiB) -> /dev/sdq
#
# Problems (1): nothing can be written until they're fixed
#   - vtbd0p3: the source's GELI key slot 1 is in use; give its passphrase with --key-file vtbd0p3:1=FILE, or the restored system loses that way in
#
# Notes:
#   - GELI on vtbd0p4: --key-file takes vtbd0p4, gpt/data0

# ---- partition ----
# vtbd0 -> /dev/sdq: the source's table, GUIDs and boot code exactly
sh -c 'bko-holt write-table layout.json vtbd0 /dev/sdq $(diskinfo /dev/sdq | cut -f3)'
gpart show sdq

# ---- assemble ----
mkdir -p -m 700 /run/bko-holt/keys
# GELI on vtbd0p4 unlocks with /etc/geli/keyfile1.key on the restored system; until that's restored, a temporary key
dd if=/dev/urandom of=/run/bko-holt/keys/vtbd0p4.key bs=64 count=1 status=none
# GELI on /dev/sdqp4 (vtbd0p4): AES-XTS 128-bit, 512-byte sectors, version 7, as the source's
geli init -a HMAC/SHA256 -e AES-XTS -l 128 -s 512 -V 7 -P -K /run/bko-holt/keys/vtbd0p4.key -B none /dev/sdqp4
geli attach -p -k /run/bko-holt/keys/vtbd0p4.key /dev/sdqp4
# /dev/sdqp4.eli is authenticated (HMAC/SHA256): every sector's checksum is written once, all 1 GiB of it (about a minute at 100 MB/s)
sh -c 'n=$(diskinfo /dev/sdqp4.eli | cut -f3); m=$((n / 1048576)); dd if=/dev/zero of=/dev/sdqp4.eli bs=1m count=$m status=none && r=$(((n - m * 1048576) / 512)) && { [ $r -eq 0 ] || dd if=/dev/zero of=/dev/sdqp4.eli bs=512 seek=$((m * 1048576 / 512)) count=$r status=none; }'

# ---- format ----
# vtbd0p1: FAT, as the source's
newfs_msdos -F 32 -I 0xBC000001 -c 1 -m 0xf0 /dev/sdqp1
# vtbd0p3.eli: UFS as the source's (dumpfs -m); dumpfs's -s 28819448 is in 512-byte units, and newfs counts this device's 4096-byte sectors: -s 3602431
newfs -O 2 -U -a 32 -b 32768 -d 32768 -e 4096 -f 4096 -g 16384 -h 64 -i 8192 -j -k 6400 -m 8 -o time -s 3602431 /dev/sdqp3.eli
# vtbd0p4.eli: UFS as the source's (dumpfs -m)
newfs -O 2 -U -a 32 -b 32768 -d 32768 -e 4096 -f 4096 -g 16384 -h 64 -i 8192 -k 1312 -m 8 -o time -s 1050616 /dev/sdqp4.eli

# ---- mount ----
mkdir -p /mnt/bko-target
mount -t ufs -o rw /dev/sdqp3.eli /mnt/bko-target
mkdir -p /mnt/bko-target/data
mount -t ufs -o rw /dev/sdqp4.eli /mnt/bko-target/data
mkdir -p /mnt/bko-target/boot/efi
mount -t msdosfs -o rw /dev/sdqp1 /mnt/bko-target/boot/efi

# ---- restore ----
mkdir -p -m 700 /run/bko-holt
# the restore errors msdosfs's mount roots give, and only those, let pass
cat > /run/bko-holt/restore.fatroots <<'EOF'
[WARN] restore boot/efi: setting UID/GID failed.
[WARN] restore boot/efi: setting User/Group failed.
[WARN] restore boot/efi: chmod failed.
[WARN] restore boot/efi: setting file times failed.
EOF
# snapshot latest of /srv/bko-repo, with BackOtter's restore flags
RUSTIC_STRICT_EXIT=true sh -c '"$@" 2>/run/bko-holt/restore.err; s=$?; cat /run/bko-holt/restore.err >&2; [ $s -eq 1 ] || exit $s; e=$(printf '\''\033'\''); sed -e "s/$e\[[0-9;]*m//g" -e '\''s/[[:space:]]*$//'\'' /run/bko-holt/restore.err > /run/bko-holt/restore.err.plain; n=$(sed -n '\''s/^error: There were \([0-9][0-9]*\) errors.*/\1/p'\'' /run/bko-holt/restore.err.plain); k=$(grep -cxF -f /run/bko-holt/restore.fatroots /run/bko-holt/restore.err.plain); [ -n "$n" ] && [ "$n" = "$k" ] || exit 1; echo '\''bko-holt: the restore set no owner, mode or times on the root of /boot/efi: msdosfs keeps none there'\'' >&2' sh rustic-bko --repository /srv/bko-repo --no-progress restore --numeric-id --glob '!/.bko-holt/system.json' --glob '!/.bko-holt' latest /mnt/bko-target

# ---- boot ----
# the boot code (MBR, freebsd-boot) was written with the partition table
# vtbd0p3: /var/backups' GELI metadata backup, written again for the new provider (the restored one is the old provider's)
sh -c 'f=; for n in vtbd0p3; do [ ! -f /mnt/bko-target/var/backups/$n.eli ] || f=1; done; [ -z "$f" ] || { (umask 077; geli backup /dev/sdqp3 /mnt/bko-target/var/backups/sdqp3.eli) && for n in vtbd0p3; do [ $n = sdqp3 ] || rm -f /mnt/bko-target/var/backups/$n.eli; done; }'
# vtbd0p4: the restored key file /etc/geli/keyfile1.key
geli setkey -n 0 -K /mnt/bko-target/etc/geli/keyfile1.key -P /dev/sdqp4
rm -f /run/bko-holt/keys/vtbd0p4.key
# vtbd0p4: /var/backups' GELI metadata backup, written again for the new provider (the restored one is the old provider's)
sh -c 'f=; for n in vtbd0p4; do [ ! -f /mnt/bko-target/var/backups/$n.eli ] || f=1; done; [ -z "$f" ] || { (umask 077; geli backup /dev/sdqp4 /mnt/bko-target/var/backups/sdqp4.eli) && for n in vtbd0p4; do [ $n = sdqp4 ] || rm -f /mnt/bko-target/var/backups/$n.eli; done; }'
mkdir -p /mnt/bko-target/var/log/bko-holt
# this restore's log, on the restored system
cp -Rp /run/bko-holt/log/. /mnt/bko-target/var/log/bko-holt/

# ---- close ----
# everything under the target, deepest first
sh -c 'mount -p | awk '\''{ print $2 }'\'' | awk '\''$0 == "/mnt/bko-target" || index($0, "/mnt/bko-target/") == 1'\'' | sort -r | while read -r m; do umount "$m" || exit; done'
geli detach /dev/sdqp4.eli
rm -f /run/bko-holt/keys/vtbd0p4.key
