# bko-holt load plan for layout.json
# Source: host-1, FreeBSD 15.1-RELEASE-p6, UEFI boot with freebsd, dumped 2026-11-14T04:17:36Z
#
# Disks:
#   vtbd0 (16 GiB) -> /dev/sdq
#
# Notes:
#   - GELI on vtbd0p3: --key-file takes vtbd0p3, gpt/zfs0

# ---- partition ----
# vtbd0 -> /dev/sdq: the source's table, GUIDs and boot code exactly
sh -c 'bko-holt write-table layout.json vtbd0 /dev/sdq $(diskinfo /dev/sdq | cut -f3)'
gpart show sdq

# ---- assemble ----
# GELI on vtbd0p3: asks for the new passphrase, once (not echoed; kept in /run/bko-holt/keys/vtbd0p3.pass, mode 600, until it's used)
sh -c 'umask 077; mkdir -p /run/bko-holt/keys; stty -echo </dev/tty; printf '\''GELI passphrase for the new vtbd0p3: '\'' >/dev/tty; IFS= read -r p </dev/tty; s=$?; stty echo </dev/tty; printf '\''\n'\'' >/dev/tty; [ $s -eq 0 ] && [ -n "$p" ] && printf %s "$p" > /run/bko-holt/keys/vtbd0p3.pass'
# GELI on /dev/sdqp3 (vtbd0p3): AES-XTS 256-bit, 4096-byte sectors, version 7, as the source's
geli init -b -g -e AES-XTS -l 256 -s 4096 -i 3031038 -V 7 -J /run/bko-holt/keys/vtbd0p3.pass -B none /dev/sdqp3
geli attach -j /run/bko-holt/keys/vtbd0p3.pass /dev/sdqp3
# vtbd0p3's passphrase is in GELI's metadata now
rm -f /run/bko-holt/keys/vtbd0p3.pass
# ZFS pool zroot, with the source's feature flags only; its vdevs' ashift 12, as the source's
sh -c 'old=$(sysctl -n vfs.zfs.min_auto_ashift); sysctl vfs.zfs.min_auto_ashift=12 >/dev/null; zpool create -f -d -o cachefile=none -o feature@async_destroy=enabled -o feature@empty_bpobj=enabled -o feature@lz4_compress=enabled -o feature@multi_vdev_crash_dump=enabled -o feature@spacemap_histogram=enabled -o feature@enabled_txg=enabled -o feature@hole_birth=enabled -o feature@extensible_dataset=enabled -o feature@embedded_data=enabled -o feature@bookmarks=enabled -o feature@filesystem_limits=enabled -o feature@large_blocks=enabled -o feature@large_dnode=enabled -o feature@sha512=enabled -o feature@skein=enabled -o feature@edonr=enabled -o feature@userobj_accounting=enabled -o feature@encryption=enabled -o feature@project_quota=enabled -o feature@device_removal=enabled -o feature@obsolete_counts=enabled -o feature@zpool_checkpoint=enabled -o feature@spacemap_v2=enabled -o feature@allocation_classes=enabled -o feature@resilver_defer=enabled -o feature@bookmark_v2=enabled -o feature@redaction_bookmarks=enabled -o feature@redacted_datasets=enabled -o feature@bookmark_written=enabled -o feature@log_spacemap=enabled -o feature@livelist=enabled -o feature@device_rebuild=enabled -o feature@zstd_compress=enabled -o feature@draid=enabled -o feature@zilsaxattr=enabled -o feature@head_errlog=enabled -o feature@blake3=enabled -o feature@block_cloning=enabled -o feature@vdev_zaps_v2=enabled -o feature@redaction_list_spill=enabled -o feature@raidz_expansion=enabled -o feature@fast_dedup=enabled -o feature@longname=enabled -o feature@large_microzap=enabled -o feature@dynamic_gang_header=enabled -o feature@block_cloning_endian=enabled -o feature@physical_rewrite=enabled -O mountpoint=/zroot -O compression=on -O atime=off -R /mnt/bko-target zroot /dev/sdqp3.eli; s=$?; sysctl vfs.zfs.min_auto_ashift=$old >/dev/null; exit $s'
# the source pool's GUID (vdev GUIDs can't be kept)
zpool reguid -g 1000000000000000001 zroot
# mount mounts every dataset in order
sh -c '[ "$(zfs get -H -o value mounted zroot)" != yes ] || zfs unmount zroot'

# ---- format ----
# vtbd0p1: FAT, as the source's
newfs_msdos -F 32 -I 0xBC000001 -c 1 /dev/sdqp1
zfs create -u -o mountpoint=none zroot/ROOT
zfs create -u -o mountpoint=/ -o canmount=noauto zroot/ROOT/default
zfs create -u -o mountpoint=/home zroot/home
zfs create -u -o mountpoint=/tmp -o exec=on -o setuid=off zroot/tmp
zfs create -u -o mountpoint=/usr -o canmount=off zroot/usr
zfs create -u -o setuid=off zroot/usr/ports
zfs create -u zroot/usr/src
zfs create -u -o mountpoint=/var -o canmount=off zroot/var
zfs create -u -o exec=off -o setuid=off zroot/var/audit
zfs create -u -o exec=off -o setuid=off zroot/var/crash
zfs create -u -o exec=off -o setuid=off zroot/var/log
zfs create -u -o atime=on zroot/var/mail
zfs create -u -o setuid=off zroot/var/tmp
zpool set bootfs=zroot/ROOT/default zroot

# ---- mount ----
zfs mount zroot/ROOT/default
zfs mount zroot/home
zfs mount zroot/tmp
zfs mount zroot
mkdir -p /mnt/bko-target/boot/efi
mount -t msdosfs -o rw /dev/sdqp1 /mnt/bko-target/boot/efi
zfs mount zroot/usr/ports
zfs mount zroot/usr/src
zfs mount zroot/var/audit
zfs mount zroot/var/crash
zfs mount zroot/var/log
zfs mount zroot/var/mail
zfs mount zroot/var/tmp

# ---- restore ----
mkdir -p /run/bko-holt
# the restore errors msdosfs's mount roots give, and only those, let pass
cat > /run/bko-holt/restore.fatroots <<'EOF'
[WARN] restore boot/efi: setting UID/GID failed.
[WARN] restore boot/efi: setting User/Group failed.
[WARN] restore boot/efi: chmod failed.
[WARN] restore boot/efi: setting file times failed.
EOF
# snapshot latest of /srv/bko-repo, with BackOtter's restore flags
RUSTIC_STRICT_EXIT=true sh -c '"$@" 2>/run/bko-holt/restore.err; s=$?; cat /run/bko-holt/restore.err >&2; [ $s -eq 1 ] || exit $s; e=$(printf '\''\033'\''); sed -e "s/$e\[[0-9;]*m//g" -e '\''s/[[:space:]]*$//'\'' /run/bko-holt/restore.err > /run/bko-holt/restore.err.plain; n=$(sed -n '\''s/^error: There were \([0-9][0-9]*\) errors.*/\1/p'\'' /run/bko-holt/restore.err.plain); k=$(grep -cxF -f /run/bko-holt/restore.fatroots /run/bko-holt/restore.err.plain); [ -n "$n" ] && [ "$n" = "$k" ] || exit 1; echo '\''bko-holt: the restore set no owner, mode or times on the root of /boot/efi: msdosfs keeps none there'\'' >&2' sh rustic-bko --repository /srv/bko-repo --no-progress restore --numeric-id latest /mnt/bko-target

# ---- boot ----
# the boot code (MBR, freebsd-boot) was written with the partition table
# vtbd0p3: /var/backups' GELI metadata backup, written again for the new provider (the restored one is the old provider's)
sh -c 'f=; for n in vtbd0p3; do [ ! -f /mnt/bko-target/var/backups/$n.eli ] || f=1; done; [ -z "$f" ] || { (umask 077; geli backup /dev/sdqp3 /mnt/bko-target/var/backups/sdqp3.eli) && for n in vtbd0p3; do [ $n = sdqp3 ] || rm -f /mnt/bko-target/var/backups/$n.eli; done; }'
# the restored system's zpool.cache names the new pool
sh -c 'for c in /etc/zfs/zpool.cache /boot/zfs/zpool.cache; do [ -f /mnt/bko-target$c ] && zpool set cachefile=/mnt/bko-target$c zroot; done; true'
mkdir -p /mnt/bko-target/var/log/bko-holt
# this restore's log, on the restored system
cp -Rp /run/bko-holt/log/. /mnt/bko-target/var/log/bko-holt/

# ---- close ----
# everything under the target, deepest first
sh -c 'mount -p | awk '\''{ print $2 }'\'' | awk '\''$0 == "/mnt/bko-target" || index($0, "/mnt/bko-target/") == 1'\'' | sort -r | while read -r m; do umount "$m" || exit; done'
zpool export zroot
geli detach /dev/sdqp3.eli
rm -f /run/bko-holt/keys/vtbd0p3.pass
