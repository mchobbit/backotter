# ---- partition ----
# vda -> /dev/sdq: clear the old table and signatures
wipefs --all /dev/sdq
# /dev/sdq: its first 4 MiB zeroed, and its last 1 MiB (a GPT's backup, md 0.90/1.0 superblocks, ZFS's end labels)
dd if=/dev/zero of=/dev/sdq bs=1M count=4 conv=fsync
sh -c 'n=$(blockdev --getsz /dev/sdq) && dd if=/dev/zero of=/dev/sdq bs=512 seek=$((n - 2048)) count=2048 conv=fsync'
# the new partitions' old content zeroed through /dev/sdq, before its table names them (blocks of 512 bytes, at their start= sectors)
# vda2: its first 4 MiB and its last 1 MiB
dd if=/dev/zero of=/dev/sdq bs=512 seek=4096 count=8192 conv=fsync
dd if=/dev/zero of=/dev/sdq bs=512 seek=4196352 count=2048 conv=fsync
# vda3: its first 4 MiB and its last 1 MiB
dd if=/dev/zero of=/dev/sdq bs=512 seek=4198400 count=8192 conv=fsync
dd if=/dev/zero of=/dev/sdq bs=512 seek=41938944 count=2048 conv=fsync
# udev done with /dev/sdq before sfdisk takes it
udevadm settle
sfdisk --wipe always --wipe-partitions always /dev/sdq <<'EOF'
label: gpt
label-id: bc000000-0000-4000-8000-000000000001
unit: sectors
sector-size: 512
table-length: 128
first-lba: 2048
last-lba: 41943006

/dev/sdq1 : start=2048, size=2048, type=21686148-6449-6E6F-744E-656564454649, uuid=BC000000-0000-4000-8000-000000000002, name="bios"
/dev/sdq2 : start=4096, size=4194304, type=0657FD6D-A4AB-43C4-84E5-0933C84B4F4F, uuid=BC000000-0000-4000-8000-000000000003, name="swap"
/dev/sdq3 : start=4198400, size=37742592, type=0FC63DAF-8483-4772-8E79-3D69D8477DE4, uuid=BC000000-0000-4000-8000-000000000005, name="root"
EOF
udevadm settle

# ---- format ----
# vda2: swap "swap"
mkswap -U bc000000-0000-4000-8000-000000000004 -L swap --pagesize 4096 /dev/sdq2
# check: /dev/sdq2 is swap with the source's UUID, label and page size (read back by bko-holt's own probe)
# vda3: btrfs "label1"
mkfs.btrfs -f -O extref,skinny-metadata,no-holes,^raid56,^raid1c34,^zoned,^squota,free-space-tree,^block-group-tree -U bc000000-0000-4000-8000-000000000006 -L label1 --csum crc32c -n 16384 -s 4096 -d single -m dup --device-uuid bc000000-0000-4000-8000-000000000007 /dev/sdq3
# check: /dev/sdq3 is btrfs with the source's features (read back by bko-holt's own probe)
mkdir -p /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000006
# btrfs bc000000-0000-4000-8000-000000000006: its subvolumes
mount -t btrfs -o subvolid=5 /dev/sdq3 /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000006
btrfs subvolume create /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000006/@
btrfs subvolume create /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000006/@home
btrfs subvolume create /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000006/@gnu
umount /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000006

# ---- mount ----
mkdir -p /mnt/bko-target
mount -t btrfs -o subvol=@ /dev/sdq3 /mnt/bko-target

# ---- restore ----
# snapshot latest of /srv/bko-repo, with BackOtter's restore flags

# the atimes the restore set, for boot to put back after its reads

# ---- boot ----
# refuses to start unless /mnt/bko-target/etc/fstab is the restored system's (its / matches the layout's)
# the restored atimes (the restore stage lists them), to put back after this stage's reads

# the restored LVM devices file, kept
sh -c 'rm -rf /run/bko-holt/lvm-devices; [ ! -d /mnt/bko-target/etc/lvm/devices ] || { cp -a /mnt/bko-target/etc/lvm/devices /run/bko-holt/lvm-devices && touch -r /mnt/bko-target/etc/lvm /run/bko-holt/lvm-devices.parent; }'

mount --rbind --make-rslave /run /mnt/bko-target/run
# Guix System's own boot.img and core.img, restored in /boot/grub/i386-pc; the rescue's grub-bios-setup writes them to the new disk
sh -c 'd=/mnt/bko-target/boot/grub/i386-pc; [ -f "$d/boot.img" ] && [ -f "$d/core.img" ] || { echo "$d: no boot.img or core.img (the source'\''s, restored with /boot/grub) to write to /dev/sdq" >&2; exit 1; }; grub-bios-setup --directory="$d" /dev/sdq'
mkdir -p /mnt/bko-target/var/log/bko-holt
# this restore's log, on the restored system
cp -a /run/bko-holt/log/. /mnt/bko-target/var/log/bko-holt/
umount --recursive /mnt/bko-target/run
umount --recursive /mnt/bko-target/sys
umount --recursive /mnt/bko-target/proc
umount --recursive /mnt/bko-target/dev
# the restored LVM devices file, put back
sh -c '[ ! -d /run/bko-holt/lvm-devices ] || { rm -rf /mnt/bko-target/etc/lvm/devices && cp -a /run/bko-holt/lvm-devices /mnt/bko-target/etc/lvm/devices && touch -r /run/bko-holt/lvm-devices.parent /mnt/bko-target/etc/lvm; }'
# the atimes this stage's reads changed, put back
bash -c 'find /mnt/bko-target \( -path /mnt/bko-target/dev -o -path /mnt/bko-target/proc -o -path /mnt/bko-target/sys -o -path /mnt/bko-target/run \) -prune -o -printf '\''%A@\t%p\0'\'' | LC_ALL=C sort -z > /run/bko-holt/atimes.after && LC_ALL=C comm -z -23 /run/bko-holt/atimes.before /run/bko-holt/atimes.after | \
             while IFS=$'\''\t'\'' read -r -d '\'''\'' a p; do [ -L "$p" ] || [ -e "$p" ] || continue; touch -a -h -d "@$a" -- "$p" 2>/dev/null || { [ ! -L "$p" ] && lsattr -d -- "$p" 2>/dev/null | cut -d'\'' '\'' -f1 | grep -q i && { chattr -i -- "$p"; touch -a -h -d "@$a" -- "$p"; chattr +i -- "$p"; }; }; done'

# ---- close ----
umount --recursive /mnt/bko-target
# other mount namespaces' copies of the target's mounts
