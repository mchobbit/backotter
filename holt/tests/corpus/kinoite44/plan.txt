# ---- partition ----
# vda -> /dev/sdq: clear the old table and signatures
wipefs --all /dev/sdq
# /dev/sdq: its first 4 MiB zeroed, and its last 1 MiB (a GPT's backup, md 0.90/1.0 superblocks, ZFS's end labels)
dd if=/dev/zero of=/dev/sdq bs=1M count=4 conv=fsync
sh -c 'n=$(blockdev --getsz /dev/sdq) && dd if=/dev/zero of=/dev/sdq bs=512 seek=$((n - 2048)) count=2048 conv=fsync'
# the new partitions' old content zeroed through /dev/sdq, before its table names them (blocks of 512 bytes, at their start= sectors)
# vda1: its first 4 MiB and its last 1 MiB
dd if=/dev/zero of=/dev/sdq bs=512 seek=2048 count=8192 conv=fsync
dd if=/dev/zero of=/dev/sdq bs=512 seek=1228800 count=2048 conv=fsync
# vda2: its first 4 MiB and its last 1 MiB
dd if=/dev/zero of=/dev/sdq bs=512 seek=1230848 count=8192 conv=fsync
dd if=/dev/zero of=/dev/sdq bs=512 seek=5423104 count=2048 conv=fsync
# vda3: its first 4 MiB and its last 1 MiB
dd if=/dev/zero of=/dev/sdq bs=512 seek=5425152 count=8192 conv=fsync
dd if=/dev/zero of=/dev/sdq bs=512 seek=67104768 count=2048 conv=fsync
# udev done with /dev/sdq before sfdisk takes it
udevadm settle
sfdisk --wipe always --wipe-partitions always /dev/sdq <<'EOF'

label-id: bc000000-0000-4000-8000-000000000001

/dev/sdq2 : start=1230848, size=4194304, type=0FC63DAF-8483-4772-8E79-3D69D8477DE4, uuid=BC000000-0000-4000-8000-000000000003
/dev/sdq3 : start=5425152, size=61681664, type=0FC63DAF-8483-4772-8E79-3D69D8477DE4, uuid=BC000000-0000-4000-8000-000000000006

udevadm settle

# (nothing to do)

# ---- format ----
# vda1: vfat
mkfs.fat -F 32 -i BC000001 -S 512 -s 8 -f 2 -R 32 -h 2048 /dev/sdq1
# check: /dev/sdq1 is vfat with the source's FAT size, sector and cluster size (read back by bko-holt's own probe)
# vda2: ext4
mke2fs -F -q -t ext4 -O none,has_journal,ext_attr,resize_inode,dir_index,orphan_file,filetype,extent,64bit,flex_bg,metadata_csum_seed,sparse_super,large_file,huge_file,dir_nlink,extra_isize,metadata_csum -U bc000000-0000-4000-8000-000000000004 -b 4096 -I 256 -N 131072 -g 32768 -G 16 -E hash_seed=bc000000-0000-4000-8000-000000000005,desc_size=64,resize=536870912 /dev/sdq2 2097152k
tune2fs -r 26214 /dev/sdq2
# check: /dev/sdq2 is ext4 with the source's features (read back by bko-holt's own probe)
# vda3: btrfs "fedora"
mkfs.btrfs -f -O extref,skinny-metadata,no-holes,^raid56,^raid1c34,^zoned,^squota,free-space-tree,block-group-tree -U bc000000-0000-4000-8000-000000000007 -L fedora --csum crc32c -n 16384 -s 4096 -d single -m dup --device-uuid bc000000-0000-4000-8000-000000000008 /dev/sdq3
# check: /dev/sdq3 is btrfs with the source's features (read back by bko-holt's own probe)
mkdir -p /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000007
# btrfs bc000000-0000-4000-8000-000000000007: its subvolumes
mount -t btrfs -o subvolid=5 /dev/sdq3 /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000007
btrfs subvolume create /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000007/var
btrfs subvolume create /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000007/home
btrfs subvolume create /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000007/root
umount /run/bko-holt/btrfs/bc000000-0000-4000-8000-000000000007

# ---- mount ----
mkdir -p /mnt/bko-target
mount -t btrfs -o subvol=root,compress=zstd:1 /dev/sdq3 /mnt/bko-target
mkdir -p /mnt/bko-target/boot
mount -t ext4 -o defaults /dev/sdq2 /mnt/bko-target/boot
mkdir -p /mnt/bko-target/home
mount -t btrfs -o subvol=home,compress=zstd:1 /dev/sdq3 /mnt/bko-target/home
mkdir -p /mnt/bko-target/var

mkdir -p /mnt/bko-target/boot/efi
mount -t vfat -o uid=0,gid=0,umask=077,shortname=winnt /dev/sdq1 /mnt/bko-target/boot/efi

# ---- restore ----
# snapshot latest of /srv/bko-repo, with BackOtter's restore flags
RUSTIC_STRICT_EXIT=true rustic-bko --repository /srv/bko-repo --no-progress restore --numeric-id latest /mnt/bko-target
# the atimes the restore set, for boot to put back after its reads
sh -c 'mkdir -p /run/bko-holt && find /mnt/bko-target \( -path /mnt/bko-target/dev -o -path /mnt/bko-target/proc -o -path /mnt/bko-target/sys -o -path /mnt/bko-target/run \) -prune -o -printf '\''%A@\t%p\0'\'' | LC_ALL=C sort -z > /run/bko-holt/atimes.before'

# ---- boot ----
# refuses to start unless /mnt/bko-target/ostree/repo/config is there: the restored physical root, with the repository and the deployments
# the restored atimes (the restore stage lists them), to put back after this stage's reads
sh -c '[ -s /run/bko-holt/atimes.before ] || { mkdir -p /run/bko-holt && find /mnt/bko-target \( -path /mnt/bko-target/dev -o -path /mnt/bko-target/proc -o -path /mnt/bko-target/sys -o -path /mnt/bko-target/run \) -prune -o -printf '\''%A@\t%p\0'\'' | LC_ALL=C sort -z > /run/bko-holt/atimes.before; }'
# ostree: /boot, the ESP and /ostree are restored verbatim; no initramfs, no grub2-mkconfig, no grub2-install, no chroot
# each loader entry's kernel, initramfs and deployment are in the restored tree, by its boot.N
sh -c 't=$(readlink -f /mnt/bko-target) || exit 1; want=loader.1; errs=; nbad=0; f=; cur=; seen=
err() { errs="${errs}ostree: $*
"; [ "$cur" = "$f" ] || { nbad=$((nbad + 1)); cur=$f; }; }
l=$(readlink "$t/boot/loader") || { echo "ostree: $t/boot/loader isn'\''t a symlink" >&2; exit 1; }
case $l in loader.[0-9]*) n=${l#loader.} ;; *) echo "ostree: $t/boot/loader leads to $l, not loader.N" >&2; exit 1 ;; esac
[ -z "$want" ] || [ "$l" = "$want" ] || echo "ostree: warning: /boot/loader is $l here, $want in the dump: the backup was taken after a deploy the dump didn'\''t see" >&2
found=0
for e in "$t"/boot/loader/entries/ostree-*.conf; do
  [ -f "$e" ] || continue
  found=$((found + 1)); f=${e##*/}
  awk '\''$1 == "linux" { k = 1 } END { exit !k }'\'' "$e" || err "$f: no linux line"
  for p in $(awk '\''$1 == "linux" || $1 == "initrd" { print $2 }'\'' "$e"); do
    [ -f "$t/boot$p" ] || err "$f: $p isn'\''t on the restored /boot"
  done
  o=$(awk '\''$1 == "options" { for (i = 2; i <= NF; i++) if ($i ~ /^ostree=/) { sub(/^ostree=/, "", $i); print $i; exit } }'\'' "$e")
  [ -n "$o" ] || { err "$f: no ostree= in its options"; continue; }
  case $o in /ostree/boot.$n/*) ;; *) err "$f: ostree=$o, but /boot/loader is $l (boot.$n)" ;; esac
  d=$(readlink -e "$t$o") || { err "$f: ostree=$o leads nowhere in the restored tree"; continue; }
  r=${d#"$t"/ostree/deploy/}
  case $r in "$d") err "$f: ostree=$o leads to $d, outside $t/ostree/deploy" ;;
    */deploy/*/*) err "$f: ostree=$o leads to ${d#"$t"}, inside a deployment, not to one" ;;
    */*/deploy/*) err "$f: ostree=$o leads to ${d#"$t"}, not to a deployment" ;;
    ?*/deploy/?*) [ -d "$d" ] || err "$f: ostree=$o leads to ${d#"$t"}, not a directory"; seen="$seen $d" ;;
    *) err "$f: ostree=$o leads to ${d#"$t"}, not to a deployment (ostree/deploy/STATEROOT/deploy/NAME)" ;;
  esac
done
for d in "$t"/ostree/deploy/*/deploy/*; do
  [ -d "$d" ] && [ ! -L "$d" ] || continue
  case "$seen " in *" $d "*) ;; *) echo "ostree: warning: no loader entry leads to ${d#"$t"} (an orphan checkout: it boots nothing, and goes at the next ostree admin cleanup)" >&2 ;; esac
done
[ "$found" -gt 0 ] || { echo "ostree: no loader entries (boot/loader/entries/ostree-*.conf) on the restored /boot" >&2; exit 1; }
[ -z "$errs" ] || { printf %s "$errs" >&2; echo "ostree: $nbad of $found loader entries inconsistent with the restored tree (a backup taken across a deploy?)" >&2; exit 1; }
echo "ostree: $found loader entries, each with its kernel, initramfs and deployment"
'
efibootmgr --quiet --create --disk /dev/sdq --part 1 --label Fedora --loader '\EFI\fedora\shimx64.efi'

' security.selinux /mnt/bko-target/var/log /mnt/bko-target/var/log/bko-holt
# the atimes this stage's reads changed, put back
bash -c 'find /mnt/bko-target \( -path /mnt/bko-target/dev -o -path /mnt/bko-target/proc -o -path /mnt/bko-target/sys -o -path /mnt/bko-target/run \) -prune -o -printf '\''%A@\t%p\0'\'' | LC_ALL=C sort -z > /run/bko-holt/atimes.after && LC_ALL=C comm -z -23 /run/bko-holt/atimes.before /run/bko-holt/atimes.after | \
             while IFS=$'\''\t'\'' read -r -d '\'''\'' a p; do [ -L "$p" ] || [ -e "$p" ] || continue; touch -a -h -d "@$a" -- "$p" 2>/dev/null || { [ ! -L "$p" ] && lsattr -d -- "$p" 2>/dev/null | cut -d'\'' '\'' -f1 | grep -q i && { chattr -i -- "$p"; touch -a -h -d "@$a" -- "$p"; chattr +i -- "$p"; }; }; done'
