sh -c '[ "$(zfs get -H -o value mounted rpool)" != yes ] || zfs unmount rpool'

# ---- format ----
zfs create -o mountpoint=legacy -o canmount=off rpool/ROOT

zfs create -o mountpoint=legacy -o canmount=noauto -o org.opensolaris.libbe:uuid=bc000000-0000-4000-8000-000000000005 -o org.opensolaris.libbe:policy=static rpool/ROOT/omnios-r151058
sh -c '[ "$(zfs get -H -o value mounted rpool/ROOT/omnios-r151058)" != yes ] || zfs unmount rpool/ROOT/omnios-r151058'
zfs create -o mountpoint=/ -o canmount=noauto -o org.opensolaris.libbe:uuid=bc000000-0000-4000-8000-000000000006 -o org.opensolaris.libbe:policy=static rpool/ROOT/omnios-r151058y
sh -c '[ "$(zfs get -H -o value mounted rpool/ROOT/omnios-r151058y)" != yes ] || zfs unmount rpool/ROOT/omnios-r151058y'
zfs create -o mountpoint=/home rpool/home
sh -c '[ "$(zfs get -H -o value mounted rpool/home)" != yes ] || zfs unmount rpool/home'
zfs create rpool/home/omnios
sh -c '[ "$(zfs get -H -o value mounted rpool/home/omnios)" != yes ] || zfs unmount rpool/home/omnios'

zfs mount -O rpool/ROOT/omnios-r151058y

zfs mount -O rpool/home/omnios

# ---- restore ----
# snapshot latest of /srv/bko-repo, with BackOtter's restore flags
rustic-bko --repository /srv/bko-repo --no-progress restore --numeric-id latest /mnt/bko-target

# ---- boot ----
# the boot code (MBR, freebsd-boot) was written with the partition table
# the restored system's zpool.cache names the new pool
