# ---- partition ----
# vda -> /dev/sdq: clear the old table and signatures
wipefs --all /dev/sdq
sfdisk --wipe always --wipe-partitions always /dev/sdq <<'EOF'

# vdb -> /dev/sdr: clear the old table and signatures
wipefs --all /dev/sdr
sfdisk --wipe always --wipe-partitions always /dev/sdr <<'EOF'

unit: sectors
sector-size: 512

/dev/sdr1 : start=2048, size=2097152, type=fd, bootable

# vdc -> /dev/sds: clear the old table and signatures
wipefs --all /dev/sds
sfdisk --wipe always --wipe-partitions always /dev/sds <<'EOF'

label-id: 0xbc000003
unit: sectors
sector-size: 512

/dev/sds1 : start=2048, size=2097152, type=fd, bootable
/dev/sds2 : start=2099200, size=31455232, type=fd

udevadm settle

# ---- assemble ----
# md array md126 (raid1)
mdadm --create /dev/md/boot --run --assume-clean --metadata=1.2 --level=raid1 --raid-devices=3 --uuid=bc000000:00004000:80000000:00000001 --homehost=localhost.localdomain --name=boot --data-offset=2048K --bitmap=internal /dev/sdq1 /dev/sdr1 /dev/sds1
# md array md127 (raid5)
mdadm --create /dev/md/label1 --run --assume-clean --metadata=1.2 --level=raid5 --raid-devices=3 --chunk=512K --layout=left-symmetric --uuid=bc000000:00004000:80000000:00000003 --homehost=localhost.localdomain --name=label1 --data-offset=9216K --bitmap=internal /dev/sdq2 /dev/sdr2 /dev/sds2
mkdir -p /run/bko-holt/lvm
# volume group rl_host-1: its metadata, for pvcreate and vgcfgrestore
cat > /run/bko-holt/lvm/rl_host-1.vg <<'EOF'

pvcreate --devicesfile '' --devices /dev/md/label1 --uuid bkoLVM-0000-0000-0000-0000-0000-000001 --restorefile /run/bko-holt/lvm/rl_host-1.vg /dev/md/label1
vgcfgrestore --devicesfile '' --devices /dev/md/label1 --file /run/bko-holt/lvm/rl_host-1.vg rl_host-1
vgchange --devicesfile '' --devices /dev/md/label1 --activate y rl_host-1

# ---- format ----
# md126: xfs
mkfs.xfs -f -m crc=1,finobt=1,rmapbt=1,reflink=1,inobtcount=1,bigtime=1,uuid=bc000000-0000-4000-8000-000000000009 -i sparse=1,nrext64=1,size=512 -n ftype=1,size=4096 -b size=4096 -s size=512 -d size=1071644672,agsize=267911168 -l lazy-count=1,size=67108864 /dev/md/boot
# dm-0: xfs
mkfs.xfs -f -m crc=1,finobt=1,rmapbt=1,reflink=1,inobtcount=1,bigtime=1,uuid=bc000000-0000-4000-8000-00000000000a -i sparse=1,nrext64=1,size=512 -n ftype=1,size=4096 -b size=4096 -s size=512 -d size=30039605248,agsize=1877475328,sunit=1024,swidth=2048 -l lazy-count=1,size=67108864,su=32768 /dev/mapper/rl_host--1-root
# dm-1: swap
mkswap -U bc000000-0000-4000-8000-00000000000b --pagesize 4096 /dev/mapper/rl_host--1-swap

# ---- mount ----
mkdir -p /mnt/bko-target
mount -t xfs -o defaults /dev/mapper/rl_host--1-root /mnt/bko-target
mkdir -p /mnt/bko-target/boot
mount -t xfs -o defaults /dev/md/boot /mnt/bko-target/boot

# ---- restore ----
# snapshot latest of /srv/bko-repo, with BackOtter's restore flags
RUSTIC_STRICT_EXIT=true rustic-bko --repository /srv/bko-repo --no-progress restore --numeric-id latest /mnt/bko-target
# the atimes the restore set, for boot to put back after its reads
sh -c 'mkdir -p /run/bko-holt && find /mnt/bko-target \( -path /mnt/bko-target/dev -o -path /mnt/bko-target/proc -o -path /mnt/bko-target/sys -o -path /mnt/bko-target/run \) -prune -o -printf '\''%A@\t%p\0'\'' | LC_ALL=C sort -z > /run/bko-holt/atimes.before'

# ---- boot ----
# refuses to start unless /mnt/bko-target/etc/fstab is the restored system's (its / matches the layout's)
# the restored atimes (the restore stage lists them), to put back after this stage's reads
sh -c '[ -s /run/bko-holt/atimes.before ] || { mkdir -p /run/bko-holt && find /mnt/bko-target \( -path /mnt/bko-target/dev -o -path /mnt/bko-target/proc -o -path /mnt/bko-target/sys -o -path /mnt/bko-target/run \) -prune -o -printf '\''%A@\t%p\0'\'' | LC_ALL=C sort -z > /run/bko-holt/atimes.before; }'
# the restored LVM devices file, kept
sh -c 'rm -rf /run/bko-holt/lvm-devices; [ ! -d /mnt/bko-target/etc/lvm/devices ] || { cp -a /mnt/bko-target/etc/lvm/devices /run/bko-holt/lvm-devices && touch -r /mnt/bko-target/etc/lvm /run/bko-holt/lvm-devices.parent; }'
mount --rbind --make-rslave /dev /mnt/bko-target/dev
mount --rbind --make-rslave /proc /mnt/bko-target/proc
mount --rbind --make-rslave /sys /mnt/bko-target/sys
mount --rbind --make-rslave /run /mnt/bko-target/run
chroot /mnt/bko-target sh -c 'n=0; for k in /lib/modules/*; do [ -d "$k/kernel" ] || continue; dracut --force --kver "${k##*/}" || exit; n=$((n + 1)); done; [ "$n" -gt 0 ]'
chroot /mnt/bko-target grub2-install --target=i386-pc /dev/sdq
chroot /mnt/bko-target grub2-install --target=i386-pc /dev/sdr
chroot /mnt/bko-target grub2-install --target=i386-pc /dev/sds
# os-prober off: it would list the rescue system's disks
sh -c '[ ! -e /mnt/bko-target/usr/bin/os-prober ] || mount --bind /mnt/bko-target/usr/bin/true /mnt/bko-target/usr/bin/os-prober'
chroot /mnt/bko-target env LANG=C grub2-mkconfig -o /boot/grub2/grub.cfg
sh -c '! grep -q '\'' /mnt/bko-target/usr/bin/os-prober '\'' /proc/self/mountinfo || umount /mnt/bko-target/usr/bin/os-prober'
mkdir -p /mnt/bko-target/var/log/bko-holt
# this restore's log, on the restored system
cp -a /run/bko-holt/log/. /mnt/bko-target/var/log/bko-holt/
# SELinux labels for the files the boot stage wrote
chroot /mnt/bko-target sh -c '[ -f /etc/selinux/config ] || exit 0; . /etc/selinux/config; [ "${SELINUX:-disabled}" != disabled ] || exit 0; f=/etc/selinux/${SELINUXTYPE:-targeted}/contexts/files/file_contexts; [ ! -f "$f" ] || ! command -v setfiles >/dev/null || setfiles "$f" /boot /var/log/bko-holt'
umount --recursive /mnt/bko-target/run
umount --recursive /mnt/bko-target/sys
umount --recursive /mnt/bko-target/proc
umount --recursive /mnt/bko-target/dev
# the restored LVM devices file, put back
sh -c '[ ! -d /run/bko-holt/lvm-devices ] || { rm -rf /mnt/bko-target/etc/lvm/devices && cp -a /run/bko-holt/lvm-devices /mnt/bko-target/etc/lvm/devices && touch -r /run/bko-holt/lvm-devices.parent /mnt/bko-target/etc/lvm; }'
# the atimes this stage's reads changed, put back
bash -c 'find /mnt/bko-target \( -path /mnt/bko-target/dev -o -path /mnt/bko-target/proc -o -path /mnt/bko-target/sys -o -path /mnt/bko-target/run \) -prune -o -printf '\''%A@\t%p\0'\'' | LC_ALL=C sort -z > /run/bko-holt/atimes.after && LC_ALL=C comm -z -23 /run/bko-holt/atimes.before /run/bko-holt/atimes.after | \
             while IFS=$'\''\t'\'' read -r -d '\'''\'' a p; do [ -L "$p" ] || [ -e "$p" ] || continue; touch -a -h -d "@$a" -- "$p" 2>/dev/null || { [ ! -L "$p" ] && lsattr -d -- "$p" 2>/dev/null | cut -d'\'' '\'' -f1 | grep -q i && { chattr -i -- "$p"; touch -a -h -d "@$a" -- "$p"; chattr +i -- "$p"; }; }; done'

# ---- close ----
umount --recursive /mnt/bko-target
# other mount namespaces' copies of the target's mounts
sh -c 'me=$(readlink /proc/self/ns/mnt); seen=; for f in /proc/[0-9]*/mountinfo; do grep -q '\'' /mnt/bko-target[ /]'\'' "$f" 2>/dev/null || continue; p=${f#/proc/}; p=${p%/mountinfo}; ns=$(readlink /proc/$p/ns/mnt) || continue; case "$me $seen " in *"$ns "*) continue;; esac; seen="$seen $ns"; echo "unmounting /mnt/bko-target in $ns (pid $p, $(cat /proc/$p/comm))"; nsenter -t "$p" -m umount --recursive /mnt/bko-target || true; done'
sh -c 'i=0; until udevadm settle; vgchange --devicesfile '\'''\'' --devices /dev/md/label1 --activate n rl_host-1; do i=$((i + 1)); [ $i -lt 5 ] || exit 1; sleep 1; done'
sh -c 'i=0; until udevadm settle; mdadm --stop /dev/md/label1; do i=$((i + 1)); [ $i -lt 5 ] || exit 1; sleep 1; done'
sh -c 'i=0; until udevadm settle; mdadm --stop /dev/md/boot; do i=$((i + 1)); [ $i -lt 5 ] || exit 1; sleep 1; done'
